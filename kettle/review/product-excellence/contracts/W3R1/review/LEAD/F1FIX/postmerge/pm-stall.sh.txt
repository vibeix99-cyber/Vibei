#!/bin/bash
# LEAD-5 post-merge: the stall reproduction (.tmp/lead5-pm/stall.spec.ts) on :5560 (f817931) and :5478 (base 4d36939),
# A B B A, one browser at a time, no burners (the stall is the load, placed exactly).
set -u
OUT=$1; mkdir -p "$OUT"; cd /home/user/wt/w6-int/kettle || exit 9
run() {
  local port=$1 i=$2
  {
    echo "### round $i: env KETTLE_PORT=$port npx playwright test -c .tmp/lead5-pm/pw.config.ts --reporter=list --output=$OUT/pw-$port-$i"
    echo "### start $(date -u +%T) load $(cut -d' ' -f1-3 /proc/loadavg)"
    env KETTLE_PORT=$port npx playwright test -c .tmp/lead5-pm/pw.config.ts --reporter=list --output="$OUT/pw-$port-$i"
    local rc=$?
    echo "### end $(date -u +%T) exit=$rc load $(cut -d' ' -f1-3 /proc/loadavg)"; echo
  } >> "$OUT/stall-$port.log" 2>&1
}
run 5560 1; run 5478 1; run 5478 2; run 5560 2
echo "done $(date -u +%T)" > "$OUT/done.txt"
